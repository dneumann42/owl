#+TITLE: Error handling in Crow

#+begin_src nest
type Dividable A:
  / = @ A -> A

@ Dividable -> Dividable Zero Error
fun safeDivide a b:
  unless (= b 0):
    raise Zero
  unless (= a b):
    raise Error "A cannot be B"
  / a b

@ Dividable Dividable -> () Zero Error
fun caller2:
  define:
    value = try (safeDivide 10 0)
  print value
    
@ () -> ()   
fun caller:
  catch (caller2):
    Error msg:
      echo "Caller failed, " message
    Zero:
      echo "Cannot divide by zero dude..."
#+end_src
